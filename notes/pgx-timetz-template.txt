package main

import (
	"context"
	"fmt"
	"log"
	"time"

	"github.com/jackc/pgx/v5"
)

func main() {
	conn, err := pgx.Connect(context.Background(), "postgres://user:password@localhost/dbname")
	if err != nil {
		log.Fatal(err)
	}
	defer conn.Close(context.Background())

	var timetzStr string
	err = conn.QueryRow(context.Background(), "SELECT '12:34:56+02'::timetz").Scan(&timetzStr)
	if err != nil {
		log.Fatal(err)
	}

	fmt.Println("Raw timetz:", timetzStr)

	// Parse manually
	parsedTime, err := time.Parse("15:04:05-07", timetzStr) // Format: HH:MM:SS±TZ
	if err != nil {
		log.Fatal(err)
	}

	fmt.Println("Parsed time:", parsedTime.Format(time.RFC3339))
}
