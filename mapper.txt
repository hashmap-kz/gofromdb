package main

import (
	"bytes"
	"fmt"
	"log"
	"os"
	"text/template"
)

// Column represents a table column
type Column struct {
	Name string
	Type string
}

// Table represents a database table
type Table struct {
	Name    string
	Columns []Column
}

// Template for generating code
const mapperTemplate = `// Code generated by mapper-generator. DO NOT EDIT.
package {{.PackageName}}

import (
	"database/sql"
	"fmt"
)

// {{.StructName}} represents a row from the {{.TableName}} table.
type {{.StructName}} struct {
{{- range .Columns}}
	{{.FieldName}} {{.FieldType}} ` + "`db:\"{{.DBName}}\"`" + `
{{- end}}
}

// Scan populates a {{.StructName}} struct from a database row.
func (e *{{.StructName}}) Scan(rows *sql.Rows) error {
	if rows == nil {
		return fmt.Errorf("rows is nil")
	}
	return rows.Scan(
{{- range .Columns}}
		&{{if eq .FieldType "sql.NullString"}}e.{{.FieldName}}{{else}}e.{{.FieldName}}{{end}},
{{- end}}
	)
}
}
`

func toGoFieldName(columnName string) string {
	// Convert snake_case to CamelCase
	fieldName := ""
	upperNext := true
	for _, r := range columnName {
		if r == '_' {
			upperNext = true
			continue
		}
		if upperNext {
			fieldName += string(r - ('a' - 'A')) // Capitalize
			upperNext = false
		} else {
			fieldName += string(r)
		}
	}
	return fieldName
}

func toGoType(dbType string) string {
	// Map SQL data types to Go types (customize as needed)
	switch dbType {
	case "varchar", "text", "char":
		return "string"
	case "int", "integer", "bigint", "smallint":
		return "int"
	case "boolean":
		return "bool"
	case "timestamp", "date":
		return "sql.NullTime"
	default:
		return "sql.NullString" // Default to nullable string
	}
}

func generateCode(table Table, packageName string) string {
	// Prepare data for the template
	structName := toGoFieldName(table.Name)
	for i, col := range table.Columns {
		table.Columns[i].FieldName = toGoFieldName(col.Name)
		table.Columns[i].FieldType = toGoType(col.Type)
	}

	// Render the template
	data := struct {
		PackageName string
		TableName   string
		StructName  string
		Columns     []Column
	}{
		PackageName: packageName,
		TableName:   table.Name,
		StructName:  structName,
		Columns:     table.Columns,
	}

	var buf bytes.Buffer
	tmpl := template.Must(template.New("mapper").Parse(mapperTemplate))
	if err := tmpl.Execute(&buf, data); err != nil {
		log.Fatalf("Failed to generate code: %v", err)
	}
	return buf.String()
}

func main() {
	// Example table schema
	table := Table{
		Name: "users",
		Columns: []Column{
			{Name: "id", Type: "bigint"},
			{Name: "name", Type: "varchar"},
			{Name: "email", Type: "varchar"},
			{Name: "created_at", Type: "timestamp"},
			{Name: "is_active", Type: "boolean"},
		},
	}

	// Generate the code
	packageName := "models"
	code := generateCode(table, packageName)

	// Write the code to a file
	fileName := fmt.Sprintf("%s_gen.go", table.Name)
	if err := os.WriteFile(fileName, []byte(code), 0644); err != nil {
		log.Fatalf("Failed to write file: %v", err)
	}

	fmt.Printf("Code generated successfully: %s\n", fileName)
}
